(function(){
  const el=id=>document.getElementById(id);
  const nrm=v=>String(v||'').normalize('NFD').replace(/[\u0300-\u036f]/g,'')
    .toLowerCase().replace(/[^a-z0-9:\/-]+/g,' ').replace(/\s+/g,' ').trim();

  function active(t){ return !(t.cancelled || t.cancelChecked===true); }

  function extractCounts(msg){
    const text=String(msg||'');
    const adt=(text.match(/\b(\d+)\s*ADT\b/i)||[])[1];
    const chd=(text.match(/\b(\d+)\s*CHD\b/i)||[])[1];
    const inf=(text.match(/\b(\d+)\s*INF\b/i)||[])[1];
    return {
      adults:adt==null?null:Number(adt),
      children:chd==null?null:Number(chd),
      babies:inf==null?null:Number(inf)
    };
  }

  function extractDate(msg){
    const m=String(msg||'').match(/\b(\d{2})[-\/](\d{2})[-\/](\d{4})\b/);
    return m?`${m[3]}-${m[2]}-${m[1]}`:'';
  }

  function extractTime(msg){
    const m=String(msg||'').match(/\b([01]?\d|2[0-3])\s*[:hH]\s*([0-5]\d)(?::[0-5]\d)?\b/);
    return m?String(m[1]).padStart(2,'0')+':'+m[2]:'';
  }

  function scoreTour(name,msg){
    const ignored=new Set(['tour','full','day','sunset','city','de','do','da','del','la','el']);
    const words=nrm(name).split(' ').filter(w=>w.length>=3&&!ignored.has(w));
    const m=nrm(msg);
    return words.length?words.filter(w=>m.includes(w)).length/words.length:0;
  }

  function expectedCounts(s,t){
    return {
      adults:Number(t.adults ?? s.adults ?? 0),
      children:Number(t.children ?? s.children ?? 0),
      babies:Number(t.babies ?? s.babies ?? 0)
    };
  }

  function hotelMatch(s,t,msg){
    const source=nrm(t.location||s.hotel||'');
    if(!source)return null;
    const words=source.split(' ').filter(w=>w.length>=5 && !['hotel','punto','encontro','ponto','avenida','regiao','metropolitana'].includes(w));
    const m=nrm(msg);
    return words.length?words.some(w=>m.includes(w)):null;
  }

  function vcard(title,state,body){
    const icon=state==='ok'?'✓':state==='danger'?'✕':'!';
    return `<div class="validatorcheck ${state}"><strong>${icon} ${esc(title)}</strong><span>${esc(body)}</span></div>`;
  }

  window.validateMessage=function(){
    const s=db.sales.find(x=>String(x.id)===String(el('validatorSale')?.value));
    const msg=el('validatorMessage')?.value.trim();
    if(!s)return alert('Selecione uma venda.');
    if(!msg)return alert('Cole a mensagem da logística.');

    const expected=(s.tours||[]).filter(t=>active(t)&&t.date===localISO(1));
    if(!expected.length){
      el('validatorResult').className='alertbox validator-stop';
      el('validatorResult').innerHTML='<strong>✕ NÃO ENVIAR</strong>Nenhum passeio ativo desta venda está marcado para amanhã.';
      return;
    }

    const ranked=expected.map(t=>({t,score:scoreTour(t.name,msg)})).sort((a,b)=>b.score-a.score);
    const best=ranked[0];
    const exp=expectedCounts(s,best.t);
    const got=extractCounts(msg);
    const msgDate=extractDate(msg);
    const msgTime=extractTime(msg);
    const tourOk=best.score>=0.45;
    const dateOk=msgDate===best.t.date;
    const adultOk=got.adults!==null && got.adults===exp.adults;
    const childOk=got.children!==null && got.children===exp.children;
    const countsOk=adultOk&&childOk;
    const hotelOk=hotelMatch(s,best.t,msg);
    const essentialOk=tourOk&&dateOk&&countsOk&&!!msgTime&&(hotelOk!==false);

    const passengerText =
      `Esperado: ${exp.adults} ADT / ${exp.children} CHD` +
      ` — Recebido: ${got.adults===null?'não identificado':got.adults+' ADT'} / ${got.children===null?'não identificado':got.children+' CHD'}`;

    const cards=[
      vcard('Passeio',tourOk?'ok':'danger',tourOk?best.t.name:`Esperado: ${best.t.name}`),
      vcard('Data',dateOk?'ok':'danger',`Esperada: ${fmtDate(best.t.date)} — Recebida: ${msgDate?fmtDate(msgDate):'não identificada'}`),
      vcard('Passageiros',countsOk?'ok':'danger',passengerText),
      vcard('Hotel / coleta',hotelOk===true?'ok':hotelOk===false?'danger':'warn',
        hotelOk===true?'Local compatível':hotelOk===false?'Local divergente':'Não foi possível comparar'),
      vcard('Horário',msgTime?'ok':'danger',msgTime||'Não identificado')
    ];

    el('validatorResult').className='alertbox '+(essentialOk?'validator-ok-title':'validator-stop');
    el('validatorResult').innerHTML=
      `<strong>${essentialOk?'✓ MENSAGEM COMPATÍVEL':'✕ NÃO ENVIAR'}</strong>`+
      `<div>${essentialOk?'Os dados essenciais conferem.':'Há divergência em um ou mais dados essenciais.'}</div>`+
      `<div class="validator-result-stack">${cards.join('')}</div>`+
      (essentialOk?`<div class="actions"><button class="btn gold" onclick="saveValidatedTimeV29(${s.id},'${best.t.name.replace(/'/g,"\\'")}','${msgTime}')">Salvar horário</button></div>`:'');

    if(!essentialOk && navigator.vibrate) navigator.vibrate([100,60,100]);
  };

  window.saveValidatedTimeV29=function(id,tourName,time){
    const s=db.sales.find(x=>x.id===id),t=s?.tours?.find(x=>x.name===tourName);
    if(!s||!t)return;
    t.hour=time;
    addHistory(s,'Mensagem validada com sucesso; horário '+time+' salvo em '+t.name);
    save();
    alert('Horário salvo na venda.');
  };

  const btn=[...document.querySelectorAll('button')].find(b=>/analisar mensagem/i.test(b.textContent||''));
  if(btn)btn.onclick=window.validateMessage;
})();
