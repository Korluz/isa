function norm(s){return String(s||'').normalize('NFD').replace(/[\u0300-\u036f]/g,'').toLowerCase().replace(/[^\p{L}\p{N}:]/gu,' ').replace(/\s+/g,' ').trim()}
const STOPWORDS=new Set(['full','day','sunset','tour','city','de','do','da','dos','das','e','del','la','el','the']);
function usefulTokens(s){return norm(s).split(' ').filter(w=>w.length>=3&&!STOPWORDS.has(w))}
function scoreTour(name,msg){
 let tokens=usefulTokens(name),m=norm(msg);if(!tokens.length)return 0;
 let hits=tokens.filter(t=>m.includes(t)).length;
 return hits/tokens.length
}
function extractTime(msg){
 let m=msg.match(/\b(?:às?\s*)?([01]?\d|2[0-3])\s*[:hH]\s*([0-5]\d)\b/);
 if(!m)m=msg.match(/\b(?:pickup|hor[aá]rio|hora)\D{0,12}([01]?\d|2[0-3])\s*([0-5]\d)\b/i);
 return m?String(m[1]).padStart(2,'0')+':'+m[2]:''
}
function paxRecognized(msg,pax){
 let n=norm(msg);
 let patterns=[new RegExp('\\b'+pax+'\\s*(?:pax|passageiros?|adt|adultos?|pessoas?)\\b','i'),new RegExp('(?:pax|passageiros?|adt|adultos?|pessoas?)\\s*[:\\-]?\\s*'+pax+'\\b','i')];
 return patterns.some(r=>r.test(n))
}
function hotelRecognized(msg,hotel){
 if(!hotel)return null;let hs=usefulTokens(hotel).filter(x=>x.length>3),m=norm(msg);
 return hs.length?hs.some(t=>m.includes(t)):null
}

}
