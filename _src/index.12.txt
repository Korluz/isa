 function tourPax(t,s){
   return {
     adults:Number(t.adults??s.adults??0),
     children:Number(t.children??s.children??0),
     babies:Number(t.babies??s.babies??0)
   };
 }
 function calcTourCommission(t,s){
   if(t.cancelled||t.cancelChecked===true)return {total:0,rule:findCommissionRule(t.name),unmapped:false};
   const rule=findCommissionRule(t.name),p=tourPax(t,s);
   if(!rule)return {total:0,rule:null,unmapped:true,p};
   const adult=Number(rule.adult||0)*p.adults;
   const child=rule.child==null?0:Number(rule.child||0)*p.children;
   return {total:adult+child,adult,child,rule,p,unmapped:false,childUnavailable:rule.child==null&&p.children>0};
 }
 function calcSaleCommission(s){
   let total=0,details=[],unmapped=[];
   (s.tours||[]).forEach(t=>{
     const c=calcTourCommission(t,s);total+=c.total;details.push({tour:t,...c});
     if(c.unmapped)unmapped.push(t.name);
   });
   return {total,details,unmapped};
 }

 function centsBRL(v){return Math.round(Number(v||0)*100)}
 function updateCommissions(){
   db.sales.forEach(s=>{
     // V10.2.6: chama window.calcSaleCommission explicitamente (não o
     // identificador local calcSaleCommission) — patches mais tardios no
     // arquivo substituem window.calcSaleCommission por uma versão com
     // regras especiais (ex: Hop On Hop Off). Chamando o identificador
     // local, updateCommissions ficava presa pra sempre na versão
     // original, mesmo sendo chamada de novo depois desses patches.
     const calc=window.calcSaleCommission(s);
     s.commissionCents=centsBRL(calc.total);
     // V10.2.4: antes só o total da venda era gravado. A comissão de cada
     // passeio (t.commissionCents) nunca era atualizada pelo motor de
     // cálculo, então "Minhas vendas", a ficha da venda, o Financeiro e o
     // CSV sempre mostravam R$0,00 ali, mesmo quando a Auditoria (que lê
     // calcSaleCommission diretamente) já calculava o valor certo.
     calc.details.forEach(d=>{d.tour.commissionCents=centsBRL(d.total)});
   });
   localStorage.setItem(KEY,JSON.stringify(db));
 }
