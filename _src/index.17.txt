function auditRows(){
 const start=document.getElementById('auditStart')?.value||'',end=document.getElementById('auditEnd')?.value||'',q=(document.getElementById('auditSearch')?.value||'').toLowerCase(),status=document.getElementById('auditStatus')?.value||'active';
 const rows=[];
 db.sales.forEach(s=>{const calc=window.calcSaleCommission?window.calcSaleCommission(s):{details:[]};
  (calc.details||[]).forEach(d=>{const t=d.tour,cancelled=!!(t.cancelled||t.cancelChecked===true||s.status==='Cancelado');
   if(status==='active'&&cancelled||status==='cancelled'&&!cancelled)return;if(start&&t.date<start||end&&t.date>end)return;
   if(q&&!(`${s.name} ${s.voucherFile||''} ${t.name}`.toLowerCase().includes(q)))return;
   const p=d.p||{adults:t.adults??s.adults??0,children:t.children??s.children??0};
   rows.push({date:t.date,client:s.name,voucher:s.voucherFile||'',tour:t.name,adt:p.adults||0,chd:p.children||0,saleValue:Number(s.valueCents||0)/100,adtRate:d.rule?.adult??null,chdRate:d.rule?.child??null,commission:Number(d.total||0),cancelled,rule:d.rule?.name||'Sem regra'});
  });
 });
 return rows.sort((a,b)=>a.date.localeCompare(b.date)||a.client.localeCompare(b.client));
}
