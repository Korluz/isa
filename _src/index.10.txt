   return key.includes('hop on hop off') || key.includes('city tour hop on hop off') ||
          key.includes('infinitum') || key.includes('alyan') || key.includes('monteluz') ||
          !!db.catalog.find(t=>normalizeName(t.name)===key)?.ticket;
 }

 // ---- helpers ----
 function activeTours(s){return (s.tours||[]).filter(t=>!t.cancelled&&t.cancelChecked!==true)}
 function isSaleCancelled(s){return s.status==='Cancelado'||activeTours(s).length===0}
 function paymentUrgentItems(){
   const items=[];
   db.sales.forEach(s=>{
     if(isSaleCancelled(s)||paymentState(s)==='Quitado')return;
     activeTours(s).forEach(t=>{const d=daysUntil(t.date);if(d>=0&&d<=1)items.push({sale:s,tour:t})})
   });
   return items;
 }
 function cancellationItems(){
   const items=[];
   db.sales.forEach(s=>{
     if(isSaleCancelled(s))return;
     activeTours(s).forEach(t=>{const d=daysUntil(t.date);if(d>=0&&d<=2&&t.cancelChecked===null)items.push({sale:s,tour:t})})
   });
   return items;
 }
 function tomorrowItems(){
   const items=[];
   db.sales.forEach(s=>{
     if(isSaleCancelled(s))return;
     activeTours(s).filter(t=>t.date===localISO(1)).forEach(t=>items.push({sale:s,tour:t}))
   });
   return items;
 }

 window.saveDetailV24=function(id){
   const s=db.sales.find(x=>x.id===id);if(!s)return;
   (s.tours||[]).forEach((t,i)=>{
     const wasCancelled=!!(t.cancelled||t.cancelChecked===true);
     t.hour=el('dh_'+i).value;t.location=el('dl_'+i).value;t.messageSent=el('ds_'+i).value==='true';
     const nowCancelled=el('dc_'+i).value==='cancelled';t.cancelled=nowCancelled;t.cancelChecked=nowCancelled?true:false;
     if(nowCancelled&&!wasCancelled)addHistory(s,'Passeio cancelado: '+t.name);
     if(!nowCancelled&&wasCancelled)addHistory(s,'Passeio reativado: '+t.name);
   });
   if(activeTours(s).length===0)s.status='Cancelado';
   else if(activeTours(s).some(t=>t.messageSent))s.status='Horário enviado';
   else if(s.status==='Cancelado')s.status='Reserva feita';
   addHistory(s,'Informações dos passeios atualizadas');
   save();closeDetail();
 }
 window.confirmPaid=function(id){
   const s=db.sales.find(x=>x.id===id);if(!s)return;
   s.paidCents=s.valueCents||s.paidCents||0;s.paymentDate=s.paymentDate||localISO();s.status=s.status==='Aguardando pagamento'?'Pago':s.status;
   addHistory(s,'Cliente confirmou o pagamento');save();openDetail(id);
 }
 window.cancelWholeSale=function(id){
   const s=db.sales.find(x=>x.id===id);if(!s||!confirm('Cancelar todos os passeios desta venda?'))return;
   s.status='Cancelado';(s.tours||[]).forEach(t=>{t.cancelled=true;t.cancelChecked=true});addHistory(s,'Venda inteira cancelada');save();closeDetail();
 }
