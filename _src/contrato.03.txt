DataURL(file) { return await new Promise((res,rej)=>{ const r=new FileReader(); r.onload=()=>res(r.result); r.onerror=rej; r.readAsDataURL(file); }); }
function cropWhiteCanvas(canvas, padding=28) {
  const ctx = canvas.getContext('2d');
  const {width:w,height:h}=canvas;
  const img = ctx.getImageData(0,0,w,h).data;
  let minX=w,minY=h,maxX=0,maxY=0,found=false;
  for(let y=0;y<h;y+=3){
    for(let x=0;x<w;x+=3){
      const i=(y*w+x)*4, r=img[i], g=img[i+1], b=img[i+2], a=img[i+3];
      if(a>20 && !(r>245 && g>245 && b>245)) { found=true; if(x<minX)minX=x; if(y<minY)minY=y; if(x>maxX)maxX=x; if(y>maxY)maxY=y; }
    }
  }
  if(!found) return canvas;
  minX=Math.max(0,minX-padding); minY=Math.max(0,minY-padding);
  maxX=Math.min(w,maxX+padding); maxY=Math.min(h,maxY+padding);
  const cw=maxX-minX, ch=maxY-minY;
  const out=document.createElement('canvas'); out.width=cw; out.height=ch;
  out.getContext('2d').drawImage(canvas,minX,minY,cw,ch,0,0,cw,ch);
  return out;
}
async function renderPdfFile(file, maxPages=99, crop=false) {
  if(!file) return [];
  if(file.type.startsWith('image/')) return [await fileToDataURL(file)];
  const buf = await file.arrayBuffer();
  const pdf = await pdfjsLib.getDocument({data: buf}).promise;
  const imgs=[];
  for(let i=1;i<=Math.min(pdf.numPages,maxPages);i++) {
    const page = await pdf.getPage(i);
    const viewport = page.getViewport({scale: 2.4});
    const canvas = document.createElement('canvas'); canvas.width=viewport.width; canvas.height=viewport.height;
    const ctx = canvas.getContext('2d'); await page.render({canvasContext:ctx, viewport}).promise;
    const finalCanvas = crop ? cropWhiteCanvas(canvas, 34) : canvas;
    imgs.push(finalCanvas.toDataURL('image/jpeg',0.95));
  }
  return imgs;
}
function fitImage(pdf, dataUrl, x, y, maxW, maxH) {
  const props = pdf.getImageProperties(dataUrl);
  let w = maxW, h = w * props.height / props.width;
  if(h > maxH) { h = maxH; w = h * props.width / props.height; }
  pdf.addImage(dataUrl, dataUrl.includes('image/png')?'PNG':'JPEG', x, y, w, h);
  return {w,h};
}

    if(!window.pdfjsLib){
      setStatus('Erro: biblioteca PDF.js não carregou. Verifique a internet ou use Chrome/Edge.');
      alert('A biblioteca PDF.js não carregou. Esta versão usa bibliotecas online. Verifique a internet e abra no Chrome/Edge.');
      return;
    }
    if(window.pdfjsLib && pdfjsLib.GlobalWorkerOptions){
      pdfjsLib.GlobalWorkerOptions.workerSrc = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';

    console.error(err);
    setStatus('Erro ao gerar: ' + (err && err.message ? err.message : err));
    alert('Erro ao gerar PDF: ' + (err && err.message ? err.message : err));
  }
}

function setupSignaturePad(){
 const canvas=document.getElementById('signaturePad');if(!canvas)return;
 const ctx=canvas.getContext('2d');
 let drawing=false,last=null;
 function resize(){
  const ratio=Math.max(window.devicePixelRatio||1,1);
  const rect=canvas.getBoundingClientRect();
  const oldInk=window.signaturePadHasInk;

  return {x:p.clientX-rect.left,y:p.clientY-rect.top};
 }

 function move(ev){if(!drawing)return;ev.preventDefault();const p=point(ev);ctx.beginPath();ctx.moveTo(last.x,last.y);ctx.lineTo(p.x,p.y);ctx.stroke();last=p;window.signaturePadHasInk=true;const st=document.getElementById('signatureStatus');if(st)st.textContent='Assinatura capturada.'}
 function end(ev){if(drawing){ev.preventDefault();drawing=false;last=null}}
 resize();window.addEventListener('resize',resize);

 canvas.addEventListener('touchstart',start,{passive:false});canvas.addEventListener('touchmove',move,{passive:false});canvas.addEventListener('touchend',end,{passive:false});

 });
