  function cleanNorm(v){
    return String(v||'').normalize('NFD').replace(/[\u0300-\u036f]/g,'')
      .toLowerCase().replace(/[^a-z0-9:]+/g,' ').replace(/\s+/g,' ').trim();
  }
  function activeTour(t){ return !(t.cancelled || t.cancelChecked===true); }

  // 1) Telefone: procura o número diretamente no cabeçalho do PDF.
  const previousParseVoucher=window.parseVoucher;
  window.parseVoucher=function(raw){
    const v=previousParseVoucher(raw);
    const txt=String(raw||'').replace(/\u00a0/g,' ').replace(/[ \t]+/g,' ');
    const patterns=[
      /Telefone\s*:\s*(\(\s*\d{2}\s*\)\s*\d{4,5}\s*[-–]?\s*\d{4})/i,
      /Telefone\s*:\s*(\d{2}\s+\d{4,5}\s*[-–]?\s*\d{4})/i,
      /Telefone\s*:\s*(\d{10,11})/i
    ];
    for(const p of patterns){
      const m=txt.match(p);
      if(m){ v.phone=phoneMask(m[1]); break; }
    }
    return v;
  };

  // 2) City Tour Hop On Hop Off usa comissão do City Walking Tour.
  const previousCalc=window.calcSaleCommission;
  function hopOnHopOffCommission(sale,tour){
    const n=cleanNorm(tour.name);
    if(!n.includes('hop on hop off')) return null;
    const adt=Number(tour.adults ?? sale.adults ?? 0);
    const chd=Number(tour.children ?? sale.children ?? 0);
    return {total:(adt*18)+(chd*18),adult:adt*18,child:chd*18,p:{adults:adt,children:chd,babies:Number(tour.babies??sale.babies??0)},rule:{name:'Hop On Hop Off',adult:18,child:18},unmapped:false};
  }
  window.calcSaleCommission=function(sale){
    const base=previousCalc(sale);
    let changed=false,total=0;
    const details=(sale.tours||[]).map(t=>{
      let current=base.details.find(d=>d.tour===t) || {};
      const hop=activeTour(t)?hopOnHopOffCommission(sale,t):null;
      if(hop){ changed=true; current={tour:t,...hop}; }
      total+=Number(current.total||0);
      return current;
    });
    return changed?{total,details,unmapped:details.filter(d=>d.unmapped).map(d=>d.tour.name)}:base;
  };
