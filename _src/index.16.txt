async function boot(){
 if(!init())return;

 const params=new URLSearchParams(location.search);
 const hashParams=new URLSearchParams(location.hash.replace(/^#/,''));

 const openRecovery=()=>{
  E('authScreen').style.display='none';
  E('recoveryScreen').classList.add('open');
  E('recoveryScreen').setAttribute('aria-hidden','false');
  recoveryMsg('Link confirmado. Defina sua nova senha.',true);
 };

 // IMPORTANTE: o listener entra imediatamente após createClient, antes de getSession().
 // O Supabase usa PASSWORD_RECOVERY para distinguir este acesso de um login normal.
 sb.auth.onAuthStateChange((event,s)=>setTimeout(async()=>{
  if(event==='PASSWORD_RECOVERY'){
   recoveryLanding=true;
   openRecovery();
   return;
  }
  // Se a URL original já identificava recuperação, não deixa a sessão normal
  // esconder a tela de nova senha enquanto o Supabase termina de processar o link.
  if(recoveryLanding && s){
   openRecovery();
   return;
  }
  await session(s);
 },0));

 if(params.get('confirmed')==='1')msg('E-mail confirmado. Você já pode entrar.',true);

 const authError=params.get('error_description')||hashParams.get('error_description');
 if(authError){
  msg(decodeURIComponent(authError.replace(/\+/g,' ')));
 }

 const {data,error}=await sb.auth.getSession();
 if(error){
  msg('Não foi possível validar o link de acesso: '+error.message);
  return;
 }

 // Fallback: se o listener não disparar por alguma particularidade do navegador,
 // a URL original + uma sessão válida ainda levam à tela correta.
 if(recoveryLanding && data.session){
  openRecovery();
 }else{
  await session(data.session);
 }
}
